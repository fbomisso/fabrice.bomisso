-- INNER JOIN : Affiche les informations d’abonnement en associant chaque abonnement
-- à l’entreprise correspondante grâce à une jointure entre les deux tables.
SELECT 
    e.nom_entreprise,
    a.date_debut,
    a.date_fin,
    a.prix_mensuel
FROM ABONNEMENTS a
INNER JOIN ENTREPRISES e ON a.id_entreprise = e.id_entreprise;

-- Compte le nombre d'abonnements pour chaque entreprise.
-- Le LEFT JOIN permet de conserver les entreprises qui n'ont aucun abonnement.
-- GROUP BY regroupe les abonnements par entreprise afin de pouvoir les compter.
SELECT 
    e.nom_entreprise,
    COUNT(a.id_abonnement) AS nombre_abonnements
FROM ENTREPRISES e
LEFT JOIN ABONNEMENTS a ON e.id_entreprise = a.id_entreprise
GROUP BY e.id_entreprise, e.nom_entreprise;

-- Calcule le revenu mensuel total pour chaque entreprise
-- en additionnant le prix des abonnements encore actifs.
-- 'a' est l'alias de la table ABONNEMENTS et 'e' celui de ENTREPRISES.
SELECT 
    e.nom_entreprise,
    SUM(a.prix_mensuel) AS revenu_total_mensuel
FROM ABONNEMENTS a
INNER JOIN ENTREPRISES e ON a.id_entreprise = e.id_entreprise
WHERE a.date_fin IS NULL
GROUP BY e.id_entreprise, e.nom_entreprise;

-- Affiche les entreprises qui ont plus d'un utilisateur
SELECT 
    ENTREPRISES.nom_entreprise,
    COUNT(UTILISATEURS.id_utilisateur) AS nombre_utilisateurs
FROM ENTREPRISES
LEFT JOIN UTILISATEURS 
    ON ENTREPRISES.id_entreprise = UTILISATEURS.id_entreprise
GROUP BY ENTREPRISES.id_entreprise, ENTREPRISES.nom_entreprise
HAVING COUNT(UTILISATEURS.id_utilisateur) > 1;

-- Affiche le nom de chaque entreprise et le nombre d'abonnements
-- associés à chaque entreprise grâce à une sous-requête.
SELECT 
    ENTREPRISES.nom_entreprise,
    (
        SELECT COUNT(*)
        FROM ABONNEMENTS
        WHERE ABONNEMENTS.id_entreprise = ENTREPRISES.id_entreprise
    ) AS nombre_abonnements
FROM ENTREPRISES;

-- Crée une table temporaire appelée abonnements_actifs
-- contenant uniquement les abonnements qui sont encore actifs.
WITH abonnements_actifs AS (
    SELECT 
        id_entreprise,
        prix_mensuel
    FROM ABONNEMENTS
    WHERE date_fin IS NULL
)

-- Affiche le nom de chaque entreprise
-- et calcule son revenu mensuel récurrent (MRR)
-- en additionnant les prix de ses abonnements actifs.
SELECT 
    ENTREPRISES.nom_entreprise,
    SUM(abonnements_actifs.prix_mensuel) AS mrr
FROM abonnements_actifs
INNER JOIN ENTREPRISES 
    ON abonnements_actifs.id_entreprise = ENTREPRISES.id_entreprise
GROUP BY ENTREPRISES.id_entreprise, ENTREPRISES.nom_entreprise;


-- Affiche les entreprises avec leur statut d'abonnement :
-- « Actif » si l'abonnement est toujours en cours
-- et « Churné » si l'abonnement est terminé.
-- UNION permet de combiner les deux résultats.
SELECT ENTREPRISES.nom_entreprise, 'Actif' AS statut
FROM ABONNEMENTS
INNER JOIN ENTREPRISES 
    ON ABONNEMENTS.id_entreprise = ENTREPRISES.id_entreprise
WHERE ABONNEMENTS.date_fin IS NULL

UNION

SELECT ENTREPRISES.nom_entreprise, 'Churné' AS statut
FROM ABONNEMENTS
INNER JOIN ENTREPRISES 
    ON ABONNEMENTS.id_entreprise = ENTREPRISES.id_entreprise
WHERE ABONNEMENTS.date_fin IS NOT NULL;

SELECT ENTREPRISES.nom_entreprise, 'Churné' AS statut
FROM ABONNEMENTS
INNER JOIN ENTREPRISES 
    ON ABONNEMENTS.id_entreprise = ENTREPRISES.id_entreprise
WHERE ABONNEMENTS.date_fin IS NOT NULL;


-- Affiche le nom de l'utilisateur, le nom de son entreprise
-- et le nombre de requêtes effectuées.
-- Calcule également le total des requêtes de chaque entreprise
-- grâce à la fonction fenêtre SUM() OVER (PARTITION BY).
-- 'ac' est l'alias de ACTIVITE, 'u' celui de UTILISATEURS
-- et 'e' celui de ENTREPRISES.
SELECT 
    UTILISATEURS.nom_utilisateur,
    ENTREPRISES.nom_entreprise,
    ACTIVITE.nb_requetes,
    SUM(ACTIVITE.nb_requetes) 
        OVER (PARTITION BY UTILISATEURS.id_entreprise) AS total_requetes_entreprise
FROM ACTIVITE
INNER JOIN UTILISATEURS 
    ON ACTIVITE.id_utilisateur = UTILISATEURS.id_utilisateur
INNER JOIN ENTREPRISES 
    ON UTILISATEURS.id_entreprise = ENTREPRISES.id_entreprise;

-- Classe les utilisateurs de chaque entreprise selon leur nombre total de requêtes.
-- Les requêtes sont additionnées pour chaque utilisateur, puis ROW_NUMBER()
-- attribue un rang à chaque utilisateur au sein de son entreprise.
-- PARTITION BY sépare le classement entreprise par entreprise.
-- ORDER BY classe les utilisateurs du plus grand nombre de requêtes au plus petit.

SELECT 
    ENTREPRISES.nom_entreprise,
    UTILISATEURS.nom_utilisateur,
    SUM(ACTIVITE.nb_requetes) AS total_requetes,
    ROW_NUMBER() OVER (
        PARTITION BY ENTREPRISES.id_entreprise 
        ORDER BY SUM(ACTIVITE.nb_requetes) DESC
    ) AS rang
FROM ACTIVITE
INNER JOIN UTILISATEURS 
    ON ACTIVITE.id_utilisateur = UTILISATEURS.id_utilisateur
INNER JOIN ENTREPRISES 
    ON UTILISATEURS.id_entreprise = ENTREPRISES.id_entreprise
GROUP BY 
    ENTREPRISES.id_entreprise, 
    ENTREPRISES.nom_entreprise, 
    UTILISATEURS.id_utilisateur, 
    UTILISATEURS.nom_utilisateur;
